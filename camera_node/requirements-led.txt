# Optional dependencies for the NeoPixel door light.
#
# Kept out of requirements.txt on purpose: adafruit-blinka detects and binds to
# Raspberry Pi hardware at install time, so it fails on a dev laptop and would
# break `pip install -r requirements.txt` for anyone not on a Pi.
#
# Install on the camera Pi only, and only if you have a strip wired up:
#
#     source .venv/bin/activate
#     pip install -r requirements-led.txt
#
# Then enable SPI (sudo raspi-config > Interface Options > SPI) and set
# `led.enabled: true` in the camera config, or turn the door light on for this
# camera in the hub's web UI.
#
# Wiring and power notes are in config.example.yml — in particular, do not run
# eight LEDs at full white off the Pi Zero's 5V rail.

adafruit-circuitpython-neopixel-spi==1.*
adafruit-blinka==8.*
